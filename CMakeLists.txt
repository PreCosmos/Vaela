cmake_minimum_required(VERSION 3.22)
project(vela VERSION 0.3.0 LANGUAGES CXX)

include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)

set(VELA_DATADIR "${CMAKE_INSTALL_FULL_DATADIR}/vela")
# Plasma's helper that starts a D-Bus session bus when the display manager didn't.
set(PLASMA_DBUS_HELPER "${CMAKE_INSTALL_FULL_LIBEXECDIR}/plasma-dbus-run-session-if-needed"
    CACHE FILEPATH "Path to plasma-dbus-run-session-if-needed")

configure_file(session/startvela.in startvela @ONLY)
configure_file(session/vela.desktop.in vela.desktop @ONLY)

install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/startvela" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vela.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/wayland-sessions")

# Config defaults for the Vela session only; the user's own ~/.config still wins.
install(DIRECTORY config/xdg/ DESTINATION "${VELA_DATADIR}/xdg")

install(DIRECTORY lookandfeel/org.vela.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/plasma/look-and-feel")

find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus Test Widgets)
find_package(KF6GlobalAccel REQUIRED)
find_package(PlasmaActivities REQUIRED)
find_package(LayerShellQt REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
set(CMAKE_AUTOMOC ON)
add_library(vela MODULE src/plugin.cpp)
target_compile_definitions(vela PRIVATE VELA_VERSION="${PROJECT_VERSION}")
target_link_libraries(vela PRIVATE KWin::kwin Qt6::DBus KF6::GlobalAccel Plasma::Activities)
install(TARGETS vela DESTINATION ${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kwin/plugins)
add_library(vela-radial STATIC src/radialmenu.cpp)
target_link_libraries(vela-radial PUBLIC Qt6::Gui KF6::ConfigCore KF6::Service KF6::KIOGui)
add_executable(vela-shell src/shell.cpp)
target_link_libraries(vela-shell PRIVATE Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::DBus vela-radial LayerShellQt::Interface KF6::ConfigCore)
target_compile_definitions(vela-shell PRIVATE VELA_QML_PATH="${VELA_DATADIR}/Main.qml")
install(TARGETS vela-shell DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src/Main.qml src/RadialEditor.qml session/migrate-layout.js DESTINATION ${VELA_DATADIR})
install(FILES session/vela-shell.desktop DESTINATION ${VELA_DATADIR}/xdg/autostart)
include(CTest)
if(BUILD_TESTING)
    add_executable(gesture-test tests/gestures.cpp)
    target_include_directories(gesture-test PRIVATE src)
    target_link_libraries(gesture-test PRIVATE Qt6::Test Qt6::Gui)
    add_test(NAME gestures COMMAND gesture-test)
endif()
if(BUILD_TESTING)
    add_executable(vela-window-test tests/windows.cpp)
    target_link_libraries(vela-window-test PRIVATE Qt6::Quick Qt6::DBus)
endif()
if(BUILD_TESTING)
    add_executable(radial-test tests/radialmenu.cpp)
    target_include_directories(radial-test PRIVATE src)
    target_link_libraries(radial-test PRIVATE vela-radial Qt6::Test)
    add_test(NAME radial-menu COMMAND radial-test)
    set_tests_properties(radial-menu PROPERTIES ENVIRONMENT "XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/radial-test-data")
endif()
if(BUILD_TESTING)
    add_executable(radial-ui-test tests/radialui.cpp)
    target_include_directories(radial-ui-test PRIVATE src)
    target_link_libraries(radial-ui-test PRIVATE vela-radial Qt6::Quick Qt6::Test)
    target_compile_definitions(radial-ui-test PRIVATE RADIAL_EDITOR_PATH="${CMAKE_CURRENT_SOURCE_DIR}/src/RadialEditor.qml" RADIAL_PREVIEW_PATH="${CMAKE_CURRENT_BINARY_DIR}/radial-editor-preview.png")
    add_test(NAME radial-editor COMMAND radial-ui-test)
    set_tests_properties(radial-editor PROPERTIES ENVIRONMENT "QT_FORCE_STDERR_LOGGING=1")
endif()

if(BUILD_TESTING)
    add_library(vela-input-probe MODULE tests/inputprobe.cpp)
    target_link_libraries(vela-input-probe PRIVATE KWin::kwin Qt6::DBus)
    add_executable(vela-mouse-test tests/mouse.cpp)
    target_link_libraries(vela-mouse-test PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test)
endif()
